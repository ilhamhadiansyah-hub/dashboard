function sampleData(){let s=7,r=()=>(s=(s*9301+49297)%233280)/233280-.5;

 let v={USD:16480,THB:507,JPY:109.5},out=[],d=new Date();d.setDate(d.getDate()-44);

 for(let i=0;i<45;i++){d.setDate(d.getDate()+1);if([0,6].includes(d.getDay()))continue;

  v.USD+=r()*70;v.THB+=r()*2.2;v.JPY+=r()*.7;

  out.push({d:d.toISOString().slice(0,10),USD:+v.USD.toFixed(2),THB:+v.THB.toFixed(2),JPY:+v.JPY.toFixed(2)})}

 return out}

/* ---------- parsing upload ---------- */

const MON={jan:1,feb:2,mar:3,apr:4,mei:5,may:5,jun:6,jul:7,agu:8,aug:8,ags:8,sep:9,okt:10,oct:10,nov:11,des:12,dec:12};

const p2=n=>String(n).padStart(2,'0');

function pDate(v){if(v==null||v==='')return null;

 if(v instanceof Date&&!isNaN(v)){v=new Date(v.getTime()+432e5);return `${v.getFullYear()}-${p2(v.getMonth()+1)}-${p2(v.getDate())}`}

 if(typeof v==='number'&&v>20000&&v<80000){const d=new Date(Math.round((v-25569)*864e5));return d.toISOString().slice(0,10)}

 const s=String(v).trim();let m;

 if(m=s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/))return `${m[1]}-${p2(m[2])}-${p2(m[3])}`;

 if(m=s.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})/))return `${m[3]}-${p2(m[2])}-${p2(m[1])}`;

 if(m=s.match(/^(\d{1,2})\s+([A-Za-z]{3,})\.?\s+(\d{4})/)){const mo=MON[m[2].slice(0,3).toLowerCase()];if(mo)return `${m[3]}-${p2(mo)}-${p2(m[1])}`}

 const d=new Date(s);return isNaN(d)?null:`${d.getFullYear()}-${p2(d.getMonth()+1)}-${p2(d.getDate())}`}

function pNum(v){if(typeof v==='number')return isFinite(v)?v:null;if(v==null)return null;

 let s=String(v).replace(/[^\d.,-]/g,'');if(!s)return null;

 const c=s.lastIndexOf(','),d=s.lastIndexOf('.');

 if(c>-1&&d>-1)s=c>d?s.replace(/\./g,'').replace(',','.'):s.replace(/,/g,'');

 else if(c>-1)s=/,\d{1,2}$/.test(s)&&s.split(',').length===2?s.replace(',','.'):s.replace(/,/g,'');

 else if(d>-1&&(s.split('.').length>2||/^\d{1,3}\.\d{3}$/.test(s)))s=s.replace(/\./g,'');

 const n=parseFloat(s);return isFinite(n)?n:null}

function convert(js,hint){

 if(!js.length)throw new Error('File kosong.');

 const keys=Object.keys(js[0]),map={},find=re=>keys.find(k=>re.test(k.trim()));

 const dk=find(/tanggal|date|tgl/i);if(!dk)throw new Error('Kolom Tanggal tidak ditemukan.');

 const wide=CUR.filter(c=>keys.some(k=>new RegExp('^'+c+'\\b','i').test(k.trim())));

 const put=(d,c,v)=>{if(d&&v!=null){(map[d]=map[d]||{d})[c]=+v.toFixed(4)}};

 if(wide.length){const wk={};wide.forEach(c=>wk[c]=keys.find(k=>new RegExp('^'+c+'\\b','i').test(k.trim())));

  js.forEach(r=>{const d=pDate(r[dk]);wide.forEach(c=>put(d,c,pNum(r[wk[c]])))})}

 else{const ck=find(/mata uang|currency|valuta|kode/i);if(!ck&&!hint)throw new Error('Mata uang tidak terdeteksi. Pakai kolom Mata Uang, atau tulis USD/THB/JPY di judul sheet atau nama file.');

  const nk=find(/^nilai|^value|^unit/i),mk=find(/tengah|mid/i),jk=find(/jual|sell/i),bk=find(/beli|buy/i),gk=find(/kurs|rate/i);

  js.forEach(r=>{const m=String(ck?r[ck]:hint).toUpperCase().match(/USD|THB|JPY/);if(!m)return;

   let v=null;if(mk)v=pNum(r[mk]);else if(jk&&bk){const a=pNum(r[jk]),b=pNum(r[bk]);if(a!=null&&b!=null)v=(a+b)/2}else if(gk)v=pNum(r[gk]);

   const u=nk?pNum(r[nk]):1;if(v!=null&&u>1)v=v/u;put(pDate(r[dk]),m[0],v)})}

 const out=Object.values(map);if(!out.length)throw new Error('Tidak ada data USD/THB/JPY yang bisa dibaca.');return out}

function parseWorkbook(wb,fname){let out=[];

 wb.SheetNames.forEach(n=>{const aoa=XLSX.utils.sheet_to_json(wb.Sheets[n],{header:1,defval:'',raw:true});

  const hi=aoa.findIndex(r=>r.some(c=>/tanggal|date|tgl/i.test(String(c)))&&r.filter(c=>String(c).trim()!=='').length>=2);

  if(hi<0)return;

  const head=aoa[hi].map((c,i)=>String(c).trim()||('c'+i));

  const hint=((aoa.slice(0,hi).flat().join(' ')+' '+n+' '+fname).match(/USD|THB|JPY/i)||[''])[0].toUpperCase();

  const js=aoa.slice(hi+1).map(r=>Object.fromEntries(head.map((h,i)=>[h,r[i]]))).filter(o=>Object.values(o).some(v=>v!==''));

  out=out.concat(convert(js,hint))});

 if(!out.length)throw new Error('Tidak ada data USD/THB/JPY yang bisa dibaca.');return out}

function msg(t,ok){const m=$('#msg');m.textContent=t;m.className=ok?'ok':'er'}

function readFile(f){return new Promise((res,rej)=>{const csv=/\.csv$/i.test(f.name),rd=new FileReader();

 rd.onload=e=>{try{const wb=csv?XLSX.read(e.target.result,{type:'string',raw:true}):XLSX.read(e.target.result,{type:'array',cellDates:true});

  res(parseWorkbook(wb,f.name))}catch(err){rej(new Error(f.name+': '+err.message))}};

 rd.onerror=()=>rej(new Error('Gagal membaca '+f.name));

 csv?rd.readAsText(f):rd.readAsArrayBuffer(f)})}

async function handle(files){files=[...(files||[])];if(!files.length)return;

 try{let inc=[];for(const f of files)inc=inc.concat(await readFile(f));

  const base=(sample||$('#replace').checked)?{}:Object.fromEntries(rows.map(r=>[r.d,r]));

  inc.forEach(r=>{base[r.d]=Object.assign(base[r.d]||{},r)});

  rows=Object.values(base).sort((a,b)=>a.d<b.d?-1:1);sample=false;save();render();

  msg(`Berhasil: ${files.length} file, ${new Set(inc.map(r=>r.d)).size} tanggal, mata uang ${CUR.filter(c=>inc.some(r=>r[c]!=null)).join(', ')}.`,true)}

 catch(err){msg(err.message||'Gagal membaca file.',false)}

 $('#file').value=''}

$('#file').onchange=e=>handle(e.target.files);

['dragleave','drop'].forEach(t=>dz.addEventListener(t,e=>{e.preventDefault();dz.classList.remove('over')}));

dz.addEventListener('drop',e=>handle(e.dataTransfer.files));

$('#range').onchange=e=>{range=+e.target.value;render()};
